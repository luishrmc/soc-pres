  \begin{columns}[T]
    \begin{column}{.50\textwidth}
      \begin{itemize}
        \item \textbf{4 BRAMs True Dual-Port (8 KB cada):} capacidade ativa de 2048 B por janela (\texttt{0x40000000/2000} e \texttt{0x40010000/2000}).
        \item \textbf{Isolamento de portas:} Porta A atende o PS a 50 MHz; Porta B atende a PL a 100 MHz sem conten\c{c}\~ao.
        \item \textbf{Contrato em 5 etapas:} escrita MMIO $\rightarrow$ \texttt{START} $\rightarrow$ c\'omputo aut\^onomo $\rightarrow$ trava \texttt{DST\_LEN}/\texttt{DONE} $\rightarrow$ leitura exata.
      \end{itemize}
    \end{column}
