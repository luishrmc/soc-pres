\begin{frame}
  \frametitle{Pipeline \textit{Compress-then-Encrypt} e Cen\'arios de Execu\c{c}\~ao}
  \begin{columns}[T]
    \begin{column}{.48\textwidth}
      \begin{block}{Compress\~ao: LZSS (Heatshrink)}
        \begin{itemize}
          \item Janela deslizante circular de 256 B e \textit{lookahead} de 16 B \autocite{Heatshrink}.
          \item Reduz redund\^ancia antes da cifragem, preservando ganho de entropia.
          \item Casamento m\'inimo de 3 B; cabe\c{c}alho Big-Endian de 16 bits.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{block}{Cifragem: AES-128-CTR}
        \begin{itemize}
          \item Padr\~ao NIST FIPS-197 e SP 800-38A com chave de 128 bits \autocite{NISTFIPS197,NISTSP80038A}.
          \item Opera\c{c}\~ao sim\'etrica direta gera \textit{keystream} paralelo por contador.
          \item Quatro cen\'arios: C1 (SW/SW), C2 (PL/SW), C3 (SW/PL) e C4 (PL/PL).
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \fonte{README.md, ADR-017, pipeline\_mgr.h}
\end{frame}
