\begin{frame}
  \frametitle{Coprocessadores em VHDL-2008 (\texttt{lzss\_accel} e \texttt{aes\_accel})}
  \begin{columns}[T]
    \begin{column}{.48\textwidth}
      \begin{block}{\texttt{lzss\_accel} (\texttt{0x43C00000})}
        \begin{itemize}
          \item Janela deslizante de 256 B e \textit{lookahead} de 16 B (compat\'ivel com Heatshrink) \autocite{Heatshrink}.
          \item Casamento m\'inimo de 3 B; cabe\c{c}alho Big-Endian de 16 bits.
          \item \textbf{Corre\c{c}\~ao ADR-019:} estado de espera \texttt{bram\_wait\_r} para lat\^encia s\'incrona de 1 ciclo da BRAM.
          \item Verificado em VUnit/GHDL (7/7 testes).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{block}{\texttt{aes\_accel} (\texttt{0x43C10000})}
        \begin{itemize}
          \item AES-128-CTR conforme NIST FIPS-197 e SP 800-38A \autocite{NISTFIPS197,NISTSP80038A}.
          \item Pipeline iterativo de 10 rodadas a 1 rodada por ciclo (\textbf{11 ciclos/bloco} de 16 B).
          \item Expans\~ao de chave \textit{on-the-fly} e multiplica\c{c}\~ao em $GF(2^8)$ sem DSPs.
          \item Vincula \texttt{seq\_id} aos 64 bits superiores do IV para evitar reuso de \textit{keystream}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \fonte{docs/hw/LZSS\_ACCEL.md, docs/hw/AES\_ACCEL.md, ADR-019}
\end{frame}

\begin{frame}
  \frametitle{Firmware Bare-Metal C11 e Injetor ESP32}
  \begin{columns}[T]
    \begin{column}{.48\textwidth}
      \begin{itemize}
        \item \textbf{Seguran\c{c}a de barramento:} \texttt{is\_pl\_programmed()} verifica bit 2 (\texttt{PCFG\_DONE}) em \texttt{0xF800700C} antes de acessar \texttt{M\_AXI\_GP0}.
        \item \textbf{Instrumenta\c{c}\~ao PMU:} leitura direta de \texttt{PMCCNTR} a 650 MHz com subtra\c{c}\~ao calibrada de \textbf{28 ciclos}.
        \item \textbf{Refer\^encia SW:} modelos C11 est\'aticos (\texttt{lzss\_sw.c}, \texttt{aes\_ctr\_sw.c}) sem aloca\c{c}\~ao din\^amica.
      \end{itemize}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{pontoschave}[Injetor ESP32 (ESP-IDF v6.1)]
        \begin{itemize}
          \item VSPI Modo 0 a 2 MHz (suporte at\'e 25 MHz) com disparo em borda de descida no \texttt{GPIO 32} \autocite{EspressifESPIDF}.
          \item Quadro de \textbf{1035 B}: cabe\c{c}alho 8 B (\texttt{0xA5}), payload alinhado de \textbf{1024 B}, CRC-16 CCITT e EOF \texttt{0x5A}.
          \item Transmiss\~ao validada em rajadas \texttt{30/30} com zero erros de CRC ou enquadramento.
        \end{itemize}
      \end{pontoschave}
    \end{column}
  \end{columns}
  \fonte{docs/sw/PIPELINE\_MGR.md, docs/esp32/SPI\_INJECTOR.md, README.md}
\end{frame}
