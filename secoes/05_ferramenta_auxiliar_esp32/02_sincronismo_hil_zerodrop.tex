      \begin{block}{Gargalo Inicial de Captura}
        \begin{itemize}

        \end{itemize}
      \end{block}

    \begin{column}{.48\textwidth}
      \begin{pontoschave}[Protocolo \textit{Zero-Drop} Implementado]
        \begin{itemize}
          \item Pulso ativo-baixo de \textbf{250 us} em \texttt{PMOD JA1} (\texttt{Y18}) para \texttt{GPIO 32} (\textit{lockout} 50 ms, \textit{gap} 30 ms).
          \item ACK \texttt{0x81} em $< 50$ us + buffer est\'atico \texttt{s\_burst\_buf[30]}.
          \item Resultado: \textbf{60/60 quadros} com 0 erros de CRC ou enquadramento.
        \end{itemize}
      \end{pontoschave}
    \end{column}
