  \begin{columns}[T]
    \begin{column}{.50\textwidth}
      \begin{itemize}
        \item \textbf{Papel exclusivo de est\'imulo:} ESP32-WROOM (240 MHz, ESP-IDF v6.1) atua apenas como gerador de tr\'afego SPI de bancada \autocite{EspressifESPIDF}.
        \item \textbf{Dataset em flash (\texttt{seed=42}):} 30 quadros fixos de \textbf{1024 B} (64 registros de 16 B) em 3 perfis: \textit{Quiescent}, \textit{Dynamic} e \textit{Chaotic}.
        \item \textbf{Enlace VSPI:} Modo 0 a 2 MHz com suporte at\'e 25 MHz.
      \end{itemize}
    \end{column}
