\begin{frame}
  \frametitle{Topologia Heterog\^enea PS/PL e Ingest\~ao ESP32}
  \begin{columns}[T]
    \begin{column}{.48\textwidth}
      \begin{itemize}
        \item \textbf{Ingest\~ao SPI (ADR-012/016):} \texttt{axi\_quad\_spi\_0} (\texttt{0x41E00000}) no Header J7 (\texttt{H15}, \texttt{T12}, \texttt{W15}, \texttt{F16}) em Modo 0.
        \item \textbf{Trigger aut\^onomo:} pino EMIO \texttt{Y18} (PMOD JA1) pulsa \texttt{GPIO 32} do ESP32 por 30 ms.
        \item \textbf{Desacoplamento BRAM:} 4 blocos True Dual-Port de 8 KB (Porta A PS a 50 MHz; Porta B PL a 100 MHz).
        \item \textbf{Pipeline:} \textit{Compress-then-Encrypt} com \texttt{lzss\_accel} e \texttt{aes\_accel}.
      \end{itemize}
    \end{column}

    \end{column}
  \end{columns}

\begin{frame}
  \frametitle{Evolu\c{c}\~ao Arquitetural: Da Proposta Inicial ao ADR-017}
  \begin{columns}[T]
    \begin{column}{.48\textwidth}
      \begin{block}{Proposta Inicial (Superada pelo ADR-017)}
        \begin{itemize}
          \item Quatro n\'ucleos via AXI4-Stream: RLE, LZ4, AES-CTR e ChaCha20.
          \item Roteamento por \texttt{axis\_accel\_mux} e transfer\^encia via AXI DMA.
          \item Gargalo: compress\~ao gera tamanho vari\'avel, causando travamento no canal S2MM do DMA.
          \item Press\~ao de roteamento na FPGA Artix-7 (17.600 LUTs).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{pontoschave}[Arquitetura Final (ADR-012 / ADR-017)]
        \begin{itemize}
          \item Substitui\c{c}\~ao do AXI DMA por \textbf{BRAM True Dual-Port} com c\'opia MMIO controlada pelo PS.
          \item Padroniza\c{c}\~ao em \textbf{LZSS + AES-128-CTR} (\textit{Compress-then-Encrypt}).
          \item Acelerador grava comprimento exato em CSR antes de sinalizar \texttt{DONE}.
          \item Contadores de ciclo internos a 100 MHz isolam lat\^encia de c\'omputo e barramento.
        \end{itemize}
      \end{pontoschave}
    \end{column}
  \end{columns}
  \fonte{utils/memory/decisions.md (ADR-012, ADR-017), TEMA.md}
\end{frame}
