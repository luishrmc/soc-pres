\begin{frame}
  \frametitle{Modelos de Refer\^encia em Software (\texttt{lzss\_sw} e \texttt{aes\_ctr\_sw})}
  \begin{columns}[T]
    \begin{column}{.48\textwidth}
      \begin{block}{\texttt{lzss\_sw.c} (Heatshrink C11)}
        \begin{itemize}
          \item Janela de 256 B (\texttt{max\_match=258}) em buffers est\'aticos \autocite{Heatshrink}.
          \item Reside integralmente na cache L1 de dados (32 KB) a 650 MHz.
          \item Fornece descompressor para valida\c{c}\~ao \textit{round-trip} do hardware.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{block}{\texttt{aes\_ctr\_sw.c} (NIST SP 800-38A)}
        \begin{itemize}
          \item Implementa\c{c}\~ao C11 com 44 palavras de subchave expandidas \autocite{NISTFIPS197,NISTSP80038A}.
          \item Validado bit-a-bit contra vetores oficiais NIST F.5.1.
          \item Define a linha de base (Cen\'ario 1) e or\'aculo de paridade para C2, C3 e C4.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

\end{frame}
